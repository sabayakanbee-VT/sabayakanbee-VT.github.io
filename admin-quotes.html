function loadQuotes() { const list = document.getElementById('quoteList'); list.innerHTML = '
読み込み中...
'; // 古いcallbackスクリプトが残っていれば削除 const oldScript = document.getElementById('jsonpScript'); if (oldScript) oldScript.remove(); window.handleQuotesData = function(data) { const quotes = data.quotes || []; if (quotes.length === 0) { list.innerHTML = '
まだ名言が登録されていません
'; return; } list.innerHTML = quotes.reverse().map(q => `
${escapeHtml(q.quote)}
${escapeHtml(q.meta)}
`).join(''); }; const script = document.createElement('script'); script.id = 'jsonpScript'; script.src = SCRIPT_URL + '?callback=handleQuotesData&t=' + Date.now(); script.onerror = function() { list.innerHTML = '
読み込みに失敗しました
'; }; document.body.appendChild(script); }
